    class CryptAudioEngine {
        constructor() {
            this.ctx = null;
            this.ambientOsc = null;
            this.ambientGain = null;
        }
        init() {
            if (!this.ctx) {
                this.ctx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (this.ctx.state === 'suspended') this.ctx.resume();
        }
        playMove(type = 'move') {
            if (!currentUser?.settings?.pieceSounds) return;
            this.init();
            const now = this.ctx.currentTime;
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            const pack = currentUser?.soundPack || 'stone';

            if (type === 'move') {
                osc.type = pack === 'metal' ? 'triangle' : 'sine';
                osc.frequency.setValueAtTime(pack === 'mystic' ? 320 : 160, now);
                osc.frequency.exponentialRampToValueAtTime(45, now + 0.09);
                gain.gain.setValueAtTime(0.25, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(now); osc.stop(now + 0.09);
            } else if (type === 'capture') {
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(280, now);
                osc.frequency.exponentialRampToValueAtTime(50, now + 0.18);
                gain.gain.setValueAtTime(0.35, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.18);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(now); osc.stop(now + 0.18);
            } else if (type === 'check') {
                osc.type = 'square';
                osc.frequency.setValueAtTime(440, now);
                osc.frequency.setValueAtTime(580, now + 0.08);
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.25);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(now); osc.stop(now + 0.25);
            } else if (type === 'blow') {
                // Dźwięk dmuchnięcia pionka w warcabach
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(500, now);
                osc.frequency.exponentialRampToValueAtTime(60, now + 0.45);
                gain.gain.setValueAtTime(0.35, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.45);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(now); osc.stop(now + 0.45);
            } else if (type === 'tick') {
                if (!currentUser?.settings?.lowTimeTick) return;
                osc.type = 'sine';
                osc.frequency.setValueAtTime(880, now);
                gain.gain.setValueAtTime(0.1, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.04);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(now); osc.stop(now + 0.04);
            }
        }
        toggleAmbient(enable) {
            if (!enable) {
                if (this.ambientOsc) { try { this.ambientOsc.stop(); } catch(e){} this.ambientOsc = null; }
                return;
            }
            this.init();
            if (this.ambientOsc) return;
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(55, this.ctx.currentTime); // Niski mroczny pomruk A1
            gain.gain.setValueAtTime(0.03, this.ctx.currentTime);
            osc.connect(gain); gain.connect(this.ctx.destination);
            osc.start();
            this.ambientOsc = osc;
            this.ambientGain = gain;
        }
    }

    function handleAvatarUpload(event, target) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            const img = new Image();
            img.onload = function() {
                const canvas = document.createElement('canvas');
                canvas.width = 128; canvas.height = 128;
                const ctx = canvas.getContext('2d');
                ctx.drawImage(img, 0, 0, 128, 128);
                const dataUrl = canvas.toDataURL('image/jpeg', 0.85);
                if (target === 'reg') {
                    selectedRegAvatar = dataUrl;
                    showToast('Wgrano własny portret!', 'success');
                } else if (currentUser) {
                    currentUser.avatar = dataUrl;
                    saveUserToStorage();
                    showToast('Zaktualizowano zdjęcie profilowe!', 'success');
                }
            };
            img.src = e.target.result;
        };
        reader.readAsDataURL(file);
    }

    function handleAuthSubmit(e) {
        e.preventDefault();
        const username = document.getElementById('authUsername').value.trim();
        const password = document.getElementById('authPassword').value.trim();
        const users = JSON.parse(localStorage.getItem('crypt_users') || '{}');

        if (authMode === 'register') {
            if (users[username]) {
                showToast('Taki wojownik już istnieje w krypcie!', 'error');
                return;
            }
            currentUser = createNewUserObject(username, password, selectedRegAvatar);
            saveUserToStorage();
            document.getElementById('authModal').classList.add('hidden');
            showToast(`Witaj w krypcie, ${username}! Otrzymujesz 150 Odłamków na start.`, 'success');
            onUserLoggedIn();
        } else {
            if (!users[username] || users[username].password !== password) {
                showToast('Błędna nazwa wojownika lub hasło!', 'error');
                return;
            }
            currentUser = users[username];
            saveUserToStorage();
            document.getElementById('authModal').classList.add('hidden');
            showToast(`Ponownie w krypcie, ${username}!`, 'success');
            onUserLoggedIn();
        }
    }

    function evaluateChessBoard(gameObj) {
        const board = gameObj.board();
        let total = 0;
        for (let r = 0; r < 8; r++) {
            for (let c = 0; c < 8; c++) {
                const piece = board[r][c];
                if (!piece) continue;
                const val = PIECE_VALUES[piece.type] + (piece.color === 'w' ? PST[piece.type][r][c] : PST[piece.type][7 - r][c]);
                total += piece.color === 'w' ? val : -val;
            }
        }
        return total;
    }

    function minimaxChess(gameObj, depth, alpha, beta, isMaximizing) {
        if (depth === 0 || gameObj.game_over()) {
            if (gameObj.in_checkmate()) return isMaximizing ? -99999 : 99999;
            return evaluateChessBoard(gameObj);
        }
        const moves = gameObj.moves({ verbose: true });
        // Sortowanie ruchów: bicia najpierw (przyśpiesza Alpha-Beta)
        moves.sort((a, b) => (b.captured ? 10 : 0) - (a.captured ? 10 : 0));

        if (isMaximizing) {
            let maxEval = -Infinity;
            for (const m of moves) {
                gameObj.move(m);
                const ev = minimaxChess(gameObj, depth - 1, alpha, beta, false);
                gameObj.undo();
                maxEval = Math.max(maxEval, ev);
                alpha = Math.max(alpha, ev);
                if (beta <= alpha) break;
            }
            return maxEval;
        } else {
            let minEval = Infinity;
            for (const m of moves) {
                gameObj.move(m);
                const ev = minimaxChess(gameObj, depth - 1, alpha, beta, true);
                gameObj.undo();
                minEval = Math.min(minEval, ev);
                beta = Math.min(beta, ev);
                if (beta <= alpha) break;
            }
            return minEval;
        }
    }

    function getBestChessMoveForElo(elo) {
        const moves = chess.moves({ verbose: true });
        if (moves.length === 0) return null;

        const botCfg = CHESS_BOTS[elo] || CHESS_BOTS[1000];
        // Celowy błąd dla niższych botów (200 - 1500 ELO)
        if (Math.random() < botCfg.mistakeRate) {
            return moves[Math.floor(Math.random() * moves.length)];
        }

        const isMaximizing = chess.turn() === 'w';
        let bestMove = moves[0];
        let bestVal = isMaximizing ? -Infinity : Infinity;
        // Ograniczamy liczbę przeszukiwanych gałęzi dla szybkości w przeglądarce przy głębokości 3-4
        const depth = elo >= 2500 ? 3 : botCfg.depth;

        moves.sort((a, b) => (b.captured ? 10 : 0) - (a.captured ? 10 : 0));

        for (const m of moves) {
            chess.move(m);
            const val = minimaxChess(chess, depth - 1, -Infinity, Infinity, !isMaximizing);
            chess.undo();
            if (isMaximizing && val > bestVal) {
                bestVal = val;
                bestMove = m;
            } else if (!isMaximizing && val < bestVal) {
                bestVal = val;
                bestMove = m;
            }
        }
        return bestMove;
    }

    /**
     * 5. SILNIK POLSKICH WARCABÓW (8x8) Z KARĄ ZA BRAK BICIA ("DMUCHNIĘCIE")
     */
    class PolishCheckersEngine {
        constructor() {
            this.reset();
        }
        reset() {
            // 8x8: null lub { color: 'w'|'b', king: boolean }
            this.board = Array.from({ length: 8 }, () => Array(8).fill(null));
            for (let r = 0; r < 8; r++) {
                for (let c = 0; c < 8; c++) {
                    if ((r + c) % 2 === 1) {
                        if (r < 3) this.board[r][c] = { color: 'b', king: false };
                        if (r > 4) this.board[r][c] = { color: 'w', king: false };
                    }
                }
            }
            this.turn = 'w';
            this.mustContinueFrom = null; // Dla wielokrotnych bić łańcuchowych
            this.comboCount = 0;
        }

        inBounds(r, c) { return r >= 0 && r < 8 && c >= 0 && c < 8; }

        // Zwraca wszystkie możliwe bicia dla danego pionka (bicie do przodu i do tyłu + latająca damka)
        getCapturesForPiece(r, c) {
            const p = this.board[r][c];
            if (!p) return [];
            const captures = [];
            const dirs = [[-1, -1], [-1, 1], [1, -1], [1, 1]];

            if (!p.king) {
                for (const [dr, dc] of dirs) {
                    const mr = r + dr, mc = c + dc;
                    const lr = r + dr * 2, lc = c + dc * 2;
                    if (this.inBounds(lr, lc) && this.board[mr][mc] && this.board[mr][mc].color !== p.color && !this.board[lr][lc]) {
                        captures.push({ fromR: r, fromC: c, toR: lr, toC: lc, capR: mr, capC: mc });
                    }
                }
            } else {
                // Damka latająca po całej przekątnej
                for (const [dr, dc] of dirs) {
                    let step = 1;
                    let foundEnemy = null;
                    while (true) {
                        const nr = r + dr * step, nc = c + dc * step;
                        if (!this.inBounds(nr, nc)) break;
                        const cell = this.board[nr][nc];
                        if (!foundEnemy) {
                            if (cell) {
                                if (cell.color === p.color) break;
                                foundEnemy = { r: nr, c: nc };
                            }
                        } else {
                            if (cell) break;
                            captures.push({ fromR: r, fromC: c, toR: nr, toC: nc, capR: foundEnemy.r, capC: foundEnemy.c });
                        }
                        step++;
                    }
                }
            }
            return captures;
        }

        getAllCapturesForColor(color) {
            if (this.mustContinueFrom) {
                return this.getCapturesForPiece(this.mustContinueFrom.r, this.mustContinueFrom.c);
            }
            let all = [];
            for (let r = 0; r < 8; r++) {
                for (let c = 0; c < 8; c++) {
                    if (this.board[r][c]?.color === color) {
                        all = all.concat(this.getCapturesForPiece(r, c));
                    }
                }
            }
            return all;
        }

        getNormalMovesForPiece(r, c) {
            if (this.mustContinueFrom) return [];
            const p = this.board[r][c];
            if (!p) return [];
            const moves = [];
            if (!p.king) {
                const forward = p.color === 'w' ? -1 : 1;
                for (const dc of [-1, 1]) {
                    const nr = r + forward, nc = c + dc;
                    if (this.inBounds(nr, nc) && !this.board[nr][nc]) {
                        moves.push({ fromR: r, fromC: c, toR: nr, toC: nc });
                    }
                }
            } else {
                const dirs = [[-1, -1], [-1, 1], [1, -1], [1, 1]];
                for (const [dr, dc] of dirs) {
                    let step = 1;
                    while (true) {
                        const nr = r + dr * step, nc = c + dc * step;
                        if (!this.inBounds(nr, nc) || this.board[nr][nc]) break;
                        moves.push({ fromR: r, fromC: c, toR: nr, toC: nc });
                        step++;
                    }
                }
            }
            return moves;
        }

        // Zwraca ruchy legalne do podświetlenia
        getUIValidMoves(r, c) {
            const p = this.board[r][c];
            if (!p || p.color !== this.turn) return [];
            if (this.mustContinueFrom && (this.mustContinueFrom.r !== r || this.mustContinueFrom.c !== c)) return [];
            return [...this.getCapturesForPiece(r, c), ...this.getNormalMovesForPiece(r, c)];
        }

        // Wykonanie ruchu z obsługą ZASADY DMUCHNIĘCIA (kara za brak bicia!)
        makeMove(fromR, fromC, toR, toC) {
            const p = this.board[fromR][fromC];
            if (!p || p.color !== this.turn) return { ok: false };

            const availableCaptures = this.getAllCapturesForColor(this.turn);
            const chosenCapture = availableCaptures.find(m => m.fromR === fromR && m.fromC === fromC && m.toR === toR && m.toC === toC);

            if (chosenCapture) {
                // Wykonujemy bicie!
                this.board[toR][toC] = p;
                this.board[fromR][fromC] = null;
                this.board[chosenCapture.capR][chosenCapture.capC] = null;
                this.comboCount++;

                // Sprawdź kolejne bicia łańcuchowe tym samym pionkiem
                const nextCaptures = this.getCapturesForPiece(toR, toC);
                if (nextCaptures.length > 0) {
                    this.mustContinueFrom = { r: toR, c: toC };
                    return { ok: true, type: 'capture_chain', combo: this.comboCount, capPos: { r: chosenCapture.capR, c: chosenCapture.capC } };
                }

                // Promocja na damkę na końcu ruchu
                if ((p.color === 'w' && toR === 0) || (p.color === 'b' && toR === 7)) p.king = true;
                const finalCombo = this.comboCount;
                this.comboCount = 0;
                this.mustContinueFrom = null;
                this.turn = this.turn === 'w' ? 'b' : 'w';
                return { ok: true, type: 'capture', combo: finalCombo, capPos: { r: chosenCapture.capR, c: chosenCapture.capC } };
            }

            // Zwykły ruch
            const normalMoves = this.getNormalMovesForPiece(fromR, fromC);
            const chosenNormal = normalMoves.find(m => m.toR === toR && m.toC === toC);
            if (!chosenNormal) return { ok: false };

            // SPRAWDZENIE KARY ZA BRAK BICIA ("DMUCHNIĘCIE")!
            let blownSquare = null;
            if (availableCaptures.length > 0) {
                // Gracz zignorował bicie! Pionek, który mógł bić, zostaje ZDMUCHNIĘTY!
                const culprit = availableCaptures[0];
                if (culprit.fromR === fromR && culprit.fromC === fromC) {
                    // Zdmuchujemy pionka z pola docelowego
                    this.board[fromR][fromC] = null;
                    blownSquare = { r: toR, c: toC };
                } else {
                    // Wykonujemy ruch, ale zdmuchujemy innego pionka, który stał i miał bicie
                    this.board[toR][toC] = p;
                    this.board[fromR][fromC] = null;
                    if ((p.color === 'w' && toR === 0) || (p.color === 'b' && toR === 7)) p.king = true;
                    this.board[culprit.fromR][culprit.fromC] = null;
                    blownSquare = { r: culprit.fromR, c: culprit.fromC };
                }
                this.turn = this.turn === 'w' ? 'b' : 'w';
                return { ok: true, type: 'blown', blownSquare };
            }

            // Czysty zwykły ruch
            this.board[toR][toC] = p;
            this.board[fromR][fromC] = null;
            if ((p.color === 'w' && toR === 0) || (p.color === 'b' && toR === 7)) p.king = true;
            this.turn = this.turn === 'w' ? 'b' : 'w';
            return { ok: true, type: 'move' };
        }

        checkWinner() {
            let wCount = 0, bCount = 0, wMoves = 0, bMoves = 0;
            for (let r = 0; r < 8; r++) {
                for (let c = 0; c < 8; c++) {
                    const p = this.board[r][c];
                    if (!p) continue;
                    if (p.color === 'w') {
                        wCount++;
                        wMoves += this.getCapturesForPiece(r, c).length + this.getNormalMovesForPiece(r, c).length;
                    } else {
                        bCount++;
                        bMoves += this.getCapturesForPiece(r, c).length + this.getNormalMovesForPiece(r, c).length;
                    }
                }
            }
            if (wCount === 0 || (this.turn === 'w' && wMoves === 0)) return 'b';
            if (bCount === 0 || (this.turn === 'b' && bMoves === 0)) return 'w';
            return null;
        }
    }

    const RUNE_FILES = ['ᚠ', 'ᚢ', 'ᚦ', 'ᚨ', 'ᚱ', 'ᚲ', 'ᚷ', 'ᚹ'];

    function coordsToAlg(r, c) {
        return String.fromCharCode(97 + c) + (8 - r);
    }
    function algToCoords(alg) {
        return { r: 8 - parseInt(alg[1]), c: alg.charCodeAt(0) - 97 };
    }

    function renderBoard() {
        const boardEl = document.getElementById('gameBoard');
        boardEl.innerHTML = '';

        const useRunes = currentUser?.settings?.runicCoords;
        const showLegal = currentUser?.settings?.showLegalMoves !== false;
        const showLast = currentUser?.settings?.highlightLastMove !== false;

        let legalTargets = [];
        if (gameState.selectedSq && showLegal) {
            if (gameState.type === 'chess') {
                const alg = coordsToAlg(gameState.selectedSq.r, gameState.selectedSq.c);
                legalTargets = chess.moves({ square: alg, verbose: true }).map(m => ({ ...algToCoords(m.to), isCap: !!m.captured }));
            } else {
                legalTargets = checkers.getUIValidMoves(gameState.selectedSq.r, gameState.selectedSq.c).map(m => ({ r: m.toR, c: m.toC, isCap: m.capR !== undefined }));
            }
        }

        const inCheck = gameState.type === 'chess' && chess.in_check();
        const turnColor = gameState.type === 'chess' ? chess.turn() : checkers.turn;

        for (let i = 0; i < 64; i++) {
            const viewR = Math.floor(i / 8);
            const viewC = i % 8;
            const r = gameState.flipped ? 7 - viewR : viewR;
            const c = gameState.flipped ? 7 - viewC : viewC;

            const sq = document.createElement('div');
            const isLight = (r + c) % 2 === 0;
            sq.className = `square relative flex items-center justify-center cursor-pointer ${isLight ? 'sq-light' : 'sq-dark'}`;
            sq.dataset.r = r;
            sq.dataset.c = c;

            // Zaznaczenie aktywnego pola
            if (gameState.selectedSq && gameState.selectedSq.r === r && gameState.selectedSq.c === c) {
                sq.classList.add('sq-selected');
            }

            // Ostatni ruch
            if (showLast && gameState.lastMove) {
                if ((gameState.lastMove.fromR === r && gameState.lastMove.fromC === c) ||
                    (gameState.lastMove.toR === r && gameState.lastMove.toC === c)) {
                    sq.classList.add('sq-last-move');
                }
            }

            // Premove
            if (gameState.premove) {
                if ((gameState.premove.fromR === r && gameState.premove.fromC === c) ||
                    (gameState.premove.toR === r && gameState.premove.toC === c)) {
                    sq.classList.add('sq-premove');
                }
            }

            // Centralne pola dla Króla Wzgórza (d4, e4, d5, e5)
            if (gameState.type === 'chess' && gameState.variant === 'koth' && (r === 3 || r === 4) && (c === 3 || c === 4)) {
                sq.style.boxShadow = 'inset 0 0 12px rgba(212, 175, 55, 0.45)';
            }

            // Współrzędne
            if (viewC === 0) {
                const rankLabel = document.createElement('span');
                rankLabel.className = 'absolute top-0.5 left-1 text-[9px] font-bold opacity-60 pointer-events-none';
                rankLabel.textContent = 8 - r;
                sq.appendChild(rankLabel);
            }
            if (viewR === 7) {
                const fileLabel = document.createElement('span');
                fileLabel.className = 'absolute bottom-0.5 right-1 text-[9px] font-bold opacity-60 pointer-events-none';
                fileLabel.textContent = useRunes ? RUNE_FILES[c] : String.fromCharCode(97 + c);
                sq.appendChild(fileLabel);
            }

            // Renderowanie figury szachowej lub pionka warcabowego
            if (gameState.type === 'chess') {
                const alg = coordsToAlg(r, c);
                const piece = chess.get(alg);
                if (piece) {
                    if (inCheck && piece.type === 'k' && piece.color === turnColor && currentUser?.settings?.checkAnimations !== false) {
                        sq.classList.add('sq-check');
                    }
                    sq.insertAdjacentHTML('beforeend', CHESS_SVGS[piece.color + piece.type]);
                }
            } else {
                const cp = checkers.board[r][c];
                if (cp) {
                    const pDiv = document.createElement('div');
                    pDiv.className = `checker-piece checker-${cp.color} ${cp.king ? 'checker-king' : ''}`;
                    sq.appendChild(pDiv);
                }
            }

            // Kropka legalnego ruchu
            const targetInfo = legalTargets.find(t => t.r === r && t.c === c);
            if (targetInfo) {
                const marker = document.createElement('div');
                marker.className = targetInfo.isCap ? 'legal-capture' : 'legal-dot';
                sq.appendChild(marker);
            }

            // Obsługa kliknięcia (LPM) i rysowania strzałek (PPM)
            sq.addEventListener('click', () => handleSquareClick(r, c));
            sq.addEventListener('mousedown', (e) => handleBoardMouseDown(e, r, c));
            sq.addEventListener('mouseup', (e) => handleBoardMouseUp(e, r, c));
            sq.addEventListener('contextmenu', (e) => e.preventDefault());

            boardEl.appendChild(sq);
        }

        updateEvalBarAndGraveyard();
    }

    function drawTacticalArrow(fromR, fromC, toR, toC, color) {
        const svg = document.getElementById('arrowOverlay');
        const vr1 = gameState.flipped ? 7 - fromR : fromR;
        const vc1 = gameState.flipped ? 7 - fromC : fromC;
        const vr2 = gameState.flipped ? 7 - toR : toR;
        const vc2 = gameState.flipped ? 7 - toC : toC;

        const x1 = vc1 * 100 + 50, y1 = vr1 * 100 + 50;
        const x2 = vc2 * 100 + 50, y2 = vr2 * 100 + 50;

        if (x1 === x2 && y1 === y2) {
            // Kółko na polu
            const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            circle.setAttribute('cx', x1); circle.setAttribute('cy', y1); circle.setAttribute('r', '42');
            circle.setAttribute('fill', 'none'); circle.setAttribute('stroke', color); circle.setAttribute('stroke-width', '7');
            circle.setAttribute('opacity', '0.8');
            svg.appendChild(circle);
            return;
        }

        const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
        line.setAttribute('x1', x1); line.setAttribute('y1', y1);
        line.setAttribute('x2', x2); line.setAttribute('y2', y2);
        line.setAttribute('stroke', color); line.setAttribute('stroke-width', '14');
        line.setAttribute('stroke-linecap', 'round'); line.setAttribute('opacity', '0.8');
        svg.appendChild(line);

        const dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
        dot.setAttribute('cx', x2); dot.setAttribute('cy', y2); dot.setAttribute('r', '16');
        dot.setAttribute('fill', color); dot.setAttribute('opacity', '0.9');
        svg.appendChild(dot);
    }

    /**
     * 7. OBSŁUGA RUCHÓW GRACZA, PREMOVE, PROMOCJI I BOTÓW
     */
    function handleSquareClick(r, c) {
        clearArrows();
        if (gameState.isGameOver) return;

        const currentTurn = gameState.type === 'chess' ? chess.turn() : checkers.turn;

        // Jeśli tura bota lub rywala P2P i włączony Premove
        if (gameState.oppMode !== 'local' && currentTurn === 'b') {
            if (!currentUser?.settings?.enablePremove) return;
            if (!gameState.selectedSq) {
                gameState.selectedSq = { r, c };
            } else {
                gameState.premove = { fromR: gameState.selectedSq.r, fromC: gameState.selectedSq.c, toR: r, toC: c };
                gameState.selectedSq = null;
                showToast('Zaplanowano Premove!', 'info');
            }
            renderBoard();
            return;
        }

        // Wybór własnej figury
        if (gameState.type === 'chess') {
            const alg = coordsToAlg(r, c);
            const piece = chess.get(alg);
            if (piece && piece.color === currentTurn) {
                gameState.selectedSq = { r, c };
                renderBoard();
                return;
            }
        } else {
            const cp = checkers.board[r][c];
            if (cp && cp.color === currentTurn) {
                gameState.selectedSq = { r, c };
                renderBoard();
                return;
            }
        }

        // Wykonanie ruchu jeśli figura była zaznaczona
        if (gameState.selectedSq) {
            if (currentUser?.settings?.confirmMove) {
                if (!gameState.pendingConfirmSq || gameState.pendingConfirmSq.r !== r || gameState.pendingConfirmSq.c !== c) {
                    gameState.pendingConfirmSq = { r, c };
                    showToast('Kliknij ponownie pole docelowe, aby potwierdzić ruch.', 'info');
                    return;
                }
            }
            gameState.pendingConfirmSq = null;
            executePlayerMove(gameState.selectedSq.r, gameState.selectedSq.c, r, c);
        }
    }

    function executePlayerMove(fromR, fromC, toR, toC, promoPiece = null) {
        if (gameState.type === 'chess') {
            const fromAlg = coordsToAlg(fromR, fromC);
            const toAlg = coordsToAlg(toR, toC);
            const piece = chess.get(fromAlg);

            // Sprawdź promocję pionka
            if (piece && piece.type === 'p' && ((piece.color === 'w' && toR === 0) || (piece.color === 'b' && toR === 7)) && !promoPiece) {
                const legal = chess.moves({ square: fromAlg, verbose: true }).some(m => m.to === toAlg);
                if (legal) {
                    gameState.pendingPromoMove = { fromR, fromC, toR, toC };
                    document.getElementById('promotionModal').classList.remove('hidden');
                    document.getElementById('promotionModal').classList.add('flex');
                    return;
                }
            }

            const moveObj = chess.move({ from: fromAlg, to: toAlg, promotion: promoPiece || 'q' });
            if (!moveObj) {
                gameState.selectedSq = null;
                renderBoard();
                return;
            }

            gameState.selectedSq = null;
            gameState.lastMove = { fromR, fromC, toR, toC };
            if (moveObj.color === 'w') gameState.whiteTime += gameState.increment;
            else gameState.blackTime += gameState.increment;

            // Ocena ruchu do Mrocznego Werdyktu
            if (moveObj.captured || chess.in_check()) gameState.moveEvaluations.great++;
            else if (Math.random() < 0.15) gameState.moveEvaluations.inacc++;

            // Dźwięk i dialog bota
            if (chess.in_check()) {
                audio.playMove('check');
                triggerBotDialogue('check');
            } else if (moveObj.captured) {
                audio.playMove('capture');
                triggerBotDialogue('lost');
                currentUser.quests.weeklyCaptures++;
            } else {
                audio.playMove('move');
            }

            updateMoveHistoryUI();
            renderBoard();
            saveAutoGame();

            if (gameState.oppMode === 'p2p' && conn && conn.open) {
                conn.send({ type: 'move', fromR, fromC, toR, toC, promo: moveObj.promotion });
            }

            // Sprawdź rozwiązanie zagadki
            if (gameState.activePuzzleSolution) {
                if (moveObj.san === gameState.activePuzzleSolution || chess.in_checkmate()) {
                    showToast('Zagadka Krypty rozwiązana! +25 Odłamków!', 'success');
                    currentUser.shards += 25;
                    gameState.activePuzzleSolution = null;
                    saveUserToStorage();
                }
            }

            if (checkChessEndConditions()) return;

            if (gameState.oppMode === 'local' && currentUser?.settings?.autoFlipLocal) {
                gameState.flipped = chess.turn() === 'b';
                renderBoard();
            }

            if (gameState.oppMode === 'bot' && chess.turn() === 'b') {
                setTimeout(triggerBotTurn, 450);
            }
        } else {
            // POLSKIE WARCABY
            const res = checkers.makeMove(fromR, fromC, toR, toC);
            if (!res.ok) {
                gameState.selectedSq = null;
                renderBoard();
                return;
            }

            gameState.lastMove = { fromR, fromC, toR, toC };
            if (res.type === 'capture_chain') {
                gameState.selectedSq = { r: toR, c: toC };
                audio.playMove('capture');
                showFloatingBanner(`BICIE ŁAŃCUCHOWE! x${res.combo}`, 'text-crypt-gold');
                renderBoard();
                return;
            }

            gameState.selectedSq = null;
            if (res.type === 'blown') {
                audio.playMove('blow');
                showFloatingBanner('💨 DMUCHNIĘCIE! STRATA PIONKA ZA BRAK BICIA!', 'text-red-500');
                showToast('Zasada Polskich Warcabów: Zignorowałeś obowiązkowe bicie i straciłeś pionka!', 'error');
            } else if (res.type === 'capture') {
                audio.playMove('capture');
                currentUser.quests.weeklyCaptures += res.combo;
                if (res.combo >= 2) {
                    showFloatingBanner(`🩸 KRWAWA RZEŹ! x${res.combo}`, 'text-crypt-bloodGlow');
                }
            } else {
                audio.playMove('move');
            }

            renderBoard();
            const winner = checkers.checkWinner();
            if (winner) {
                finishGame(winner === 'w' ? 'win' : 'lose', winner === 'w' ? 'Zbiłeś wszystkie bierki wroga!' : 'Brak ruchów w warcabach');
                return;
            }

            if (gameState.oppMode === 'bot' && checkers.turn === 'b') {
                setTimeout(triggerCheckersBotTurn, 450);
            }
        }
    }

    function confirmPromotion(pieceCode) {
        document.getElementById('promotionModal').classList.add('hidden');
        document.getElementById('promotionModal').classList.remove('flex');
        if (gameState.pendingPromoMove) {
            const { fromR, fromC, toR, toC } = gameState.pendingPromoMove;
            gameState.pendingPromoMove = null;
            executePlayerMove(fromR, fromC, toR, toC, pieceCode);
        }
    }

    function triggerBotTurn() {
        if (gameState.isGameOver || chess.turn() !== 'b') return;
        const move = getBestChessMoveForElo(gameState.botElo);
        if (!move) return;

        chess.move(move);
        const fromC = algToCoords(move.from);
        const toC = algToCoords(move.to);
        gameState.lastMove = { fromR: fromC.r, fromC: fromC.c, toR: toC.r, toC: toC.c };
        gameState.blackTime += gameState.increment;

        if (chess.in_check()) {
            audio.playMove('check');
            triggerBotDialogue('check');
        } else if (move.captured) {
            audio.playMove('capture');
            triggerBotDialogue('capture');
        } else {
            audio.playMove('move');
        }

        updateMoveHistoryUI();
        renderBoard();
        saveAutoGame();

        if (checkChessEndConditions()) return;

        // Wykonaj zaplanowany Premove gracza
        if (gameState.premove) {
            const pm = gameState.premove;
            gameState.premove = null;
            setTimeout(() => executePlayerMove(pm.fromR, pm.fromC, pm.toR, pm.toC), 120);
        }
    }

    function triggerCheckersBotTurn() {
        if (gameState.isGameOver || checkers.turn !== 'b') return;
        const caps = checkers.getAllCapturesForColor('b');
        let chosen = null;
        if (caps.length > 0) {
            chosen = caps[Math.floor(Math.random() * caps.length)];
        } else {
            const allNormals = [];
            for (let r = 0; r < 8; r++) {
                for (let c = 0; c < 8; c++) {
                    if (checkers.board[r][c]?.color === 'b') {
                        allNormals.push(...checkers.getNormalMovesForPiece(r, c));
                    }
                }
            }
            if (allNormals.length > 0) chosen = allNormals[Math.floor(Math.random() * allNormals.length)];
        }

        if (chosen) {
            const res = checkers.makeMove(chosen.fromR, chosen.fromC, chosen.toR, chosen.toC);
            gameState.lastMove = { fromR: chosen.fromR, fromC: chosen.fromC, toR: chosen.toR, toC: chosen.toC };
            audio.playMove(res.type === 'capture' || res.type === 'capture_chain' ? 'capture' : 'move');
            renderBoard();
            if (res.type === 'capture_chain') {
                setTimeout(triggerCheckersBotTurn, 400);
                return;
            }
        }
        const winner = checkers.checkWinner();
        if (winner) {
            finishGame(winner === 'w' ? 'win' : 'lose', winner === 'w' ? 'Zwycięstwo w Polskich Warcabach!' : 'Przeciwnik zdominował planszę');
        }
    }

    function checkChessEndConditions() {
        // Sprawdź Król Wzgórza (King of the Hill)
        if (gameState.variant === 'koth') {
            const centerSquares = ['d4', 'e4', 'd5', 'e5'];
            for (const sq of centerSquares) {
                const p = chess.get(sq);
                if (p && p.type === 'k') {
                    finishGame(p.color === 'w' ? 'win' : 'lose', 'Król Wzgórza zdobył centrum planszy!');
                    return true;
                }
            }
        }

        if (chess.in_checkmate()) {
            const playerWon = chess.turn() === 'b';
            triggerBotDialogue(playerWon ? 'lose' : 'win');
            finishGame(playerWon ? 'win' : 'lose', playerWon ? 'Szach-Mat! Zmiażdżyłeś przeciwnika.' : 'Zostałeś zamatowany.');
            return true;
        }
        if (chess.in_draw() || chess.in_stalemate() || chess.in_threefold_repetition()) {
            finishGame('draw', 'Partia zakończona remisem (Pat lub powtórzenie pozycji).');
            return true;
        }
        return false;
    }

    /**
     * 8. ZAKOŃCZENIE PARTII, OBLICZANIE ELO, OSIĄGNIĘCIA I MROCZNY WERDYKT
     */
    function finishGame(result, reasonText, honorBonus = false) {
        if (gameState.isGameOver) return;
        gameState.isGameOver = true;
        clearInterval(gameState.timerInterval);
        localStorage.removeItem('crypt_autosave_' + currentUser.username);

        document.getElementById('gameStatusBadge').textContent = 'Zakończona';
        document.getElementById('gameStatusBadge').className = 'text-[11px] px-2 py-0.5 rounded bg-red-900/40 text-red-400 border border-red-500/30';

        // Mnożnik nagród zależny od Koronek (3 koronki = 100%, 0 koronek = 40%)
        const crownMult = gameState.oppMode === 'bot' ? [0.4, 0.6, 0.8, 1.0][gameState.crowns] : 1.0;

        let eloDelta = 0;
        let shardsEarned = 0;

        if (result === 'win') {
            eloDelta = Math.round(16 * crownMult);
            shardsEarned = Math.round((40 + (gameState.botElo / 50)) * crownMult);
            currentUser.stats.winStreak++;
            currentUser.quests.dailyWins++;
            if (gameState.type === 'chess') {
                currentUser.chessElo += eloDelta;
                currentUser.stats.chessW++;
                currentUser.eloHistory.push(currentUser.chessElo);
            } else {
                currentUser.checkersElo += eloDelta;
                currentUser.stats.checkersW++;
            }
        } else if (result === 'lose') {
            eloDelta = honorBonus ? -8 : -12;
            shardsEarned = honorBonus ? 15 : 5;
            currentUser.stats.winStreak = 0;
            if (gameState.type === 'chess') {
                currentUser.chessElo = Math.max(100, currentUser.chessElo + eloDelta);
                currentUser.stats.chessL++;
                currentUser.eloHistory.push(currentUser.chessElo);
            } else {
                currentUser.checkersElo = Math.max(100, currentUser.checkersElo + eloDelta);
                currentUser.stats.checkersL++;
            }
        } else {
            shardsEarned = 15;
            if (gameState.type === 'chess') currentUser.stats.chessD++;
        }

        currentUser.shards += shardsEarned;
        currentUser.quests.monthlyGames++;

        // Zapisz do historii partii
        const oppLabel = gameState.oppMode === 'bot' ? `${CHESS_BOTS[gameState.botElo].name} (${gameState.botElo})` : 'Gracz Krypty';
        currentUser.history.unshift({
            game: gameState.type === 'chess' ? '♟ Szachy' : '⛀ Warcaby',
            opp: oppLabel,
            result,
            eloDelta,
            date: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
        });
        if (currentUser.history.length > 15) currentUser.history.pop();

        checkAndUnlockAchievements(result);
        saveUserToStorage();

        // Pokaż modal Mrocznego Werdyktu
        if (currentUser.settings.autoVerdictModal !== false) {
            document.getElementById('verdictIcon').textContent = result === 'win' ? '👑' : (result === 'lose' ? '💀' : '⚖️');
            document.getElementById('verdictTitle').textContent = result === 'win' ? 'ZWYCIĘSTWO W KRYPCIE!' : (result === 'lose' ? 'PORAŻKA W OTCHŁANI' : 'REMIS');
            document.getElementById('verdictSubtitle').textContent = reasonText;
            document.getElementById('verdictEloDelta').textContent = (eloDelta >= 0 ? '+' : '') + eloDelta;
            document.getElementById('verdictEloDelta').className = `text-lg font-bold ${eloDelta >= 0 ? 'text-green-400' : 'text-red-400'}`;
            document.getElementById('verdictShards').textContent = `+${shardsEarned} 💀`;

            const acc = Math.min(99.2, Math.max(54.0, 82 + gameState.moveEvaluations.great * 2.5 - gameState.moveEvaluations.inacc * 4)).toFixed(1);
            document.getElementById('verdictAccuracy').textContent = acc + '%';
            document.getElementById('verdictGreat').textContent = gameState.moveEvaluations.great;
            document.getElementById('verdictInacc').textContent = gameState.moveEvaluations.inacc;
            document.getElementById('verdictBlunders').textContent = result === 'lose' ? 2 : 0;

            document.getElementById('verdictModal').classList.remove('hidden');
            document.getElementById('verdictModal').classList.add('flex');
        }
    }

    function checkAndUnlockAchievements(result) {
        const unlock = (id) => {
            if (currentUser.achievements.includes(id)) return;
            const ach = ACHIEVEMENTS_LIST.find(a => a.id === id);
            if (!ach) return;
            currentUser.achievements.push(id);
            currentUser.shards += ach.reward;
            if (ach.userTitle && !currentUser.unlockedTitles.includes(ach.userTitle)) {
                currentUser.unlockedTitles.push(ach.userTitle);
            }
            showToast(`🏆 Osiągnięcie: ${ach.title}! (+${ach.reward} Odłamków)`, 'success');
        };

        if (result === 'win') {
            unlock('first_blood');
            if (gameState.type === 'chess' && gameState.oppMode === 'bot') {
                if (gameState.botElo >= 1500) unlock('bot_slayer');
                if (gameState.botElo >= 3000) unlock('god_killer');
            }
            if (currentUser.stats.checkersW >= 5) unlock('checkers_master');
        }
        if (currentUser.inventory.length >= 3) unlock('collector');
        if (currentUser.history.length >= 10) unlock('optimist');
    }

    function generateChess960FEN() {
        const backRank = ['r', 'n', 'b', 'q', 'k', 'b', 'n', 'r'];
        for (let i = backRank.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [backRank[i], backRank[j]] = [backRank[j], backRank[i]];
        }
        const blackStr = backRank.join('');
        const whiteStr = blackStr.toUpperCase();
        return `${blackStr}/pppppppp/8/8/8/8/PPPPPPPP/${whiteStr} w - - 0 1`;
    }

    function tickGameClock() {
        if (gameState.isGameOver) return;
        const turn = gameState.type === 'chess' ? chess.turn() : checkers.turn;
        if (turn === 'w') {
            gameState.whiteTime--;
            if (gameState.whiteTime <= 10 && gameState.whiteTime > 0) audio.playMove('tick');
            if (gameState.whiteTime <= 0) {
                gameState.whiteTime = 0;
                updateClocksUI();
                finishGame('lose', 'Twój czas w klepsydrze dobiegł końca!');
                return;
            }
        } else {
            gameState.blackTime--;
            if (gameState.blackTime <= 0) {
                gameState.blackTime = 0;
                updateClocksUI();
                finishGame('win', 'Przeciwnik przekroczył limit czasu!');
                return;
            }
        }
        updateClocksUI();
    }

    function formatClock(sec) {
        if (sec <= 0 && document.getElementById('timeControlSelect').value === '0|0') return '∞';
        const m = Math.floor(sec / 60);
        const s = sec % 60;
        return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    }

    function updateMoveHistoryUI() {
        const box = document.getElementById('moveHistoryBox');
        const history = chess.history();
        if (history.length === 0) {
            box.innerHTML = '<div class="text-gray-500 italic text-center mt-8">Brak wykonanych ruchów...</div>';
            return;
        }
        let html = '';
        for (let i = 0; i < history.length; i += 2) {
            const num = Math.floor(i / 2) + 1;
            html += `<div class="flex justify-between px-2 py-1 rounded ${i % 4 === 0 ? 'bg-crypt-bg/60' : ''}">
                <span class="text-gray-500 w-8">${num}.</span>
                <span class="text-white font-semibold flex-1">${history[i]}</span>
                <span class="text-crypt-gold flex-1">${history[i + 1] || ''}</span>
            </div>`;
        }
        box.innerHTML = html;
        box.scrollTop = box.scrollHeight;
    }

    function updateEvalBarAndGraveyard() {
        const evalContainer = document.getElementById('evalBarContainer');
        evalContainer.classList.toggle('hidden', currentUser?.settings?.showEvalBar === false);

        if (gameState.type !== 'chess') {
            document.getElementById('evalBarFill').style.height = '50%';
            document.getElementById('evalScoreText').textContent = '⛀';
            document.getElementById('playerCapturedList').innerHTML = '';
            document.getElementById('oppCapturedList').innerHTML = '';
            document.getElementById('materialAdvantage').textContent = '';
            return;
        }

        const rawScore = evaluateChessBoard(chess) / 100;
        const clamped = Math.max(-10, Math.min(10, rawScore));
        const whitePercent = 50 + (clamped * 5);
        document.getElementById('evalBarFill').style.height = whitePercent + '%';
        document.getElementById('evalScoreText').textContent = (rawScore > 0 ? '+' : '') + rawScore.toFixed(1);

        // Cmentarzysko figur
        const startCounts = { p: 8, n: 2, b: 2, r: 2, q: 1 };
        const currentW = { p: 0, n: 0, b: 0, r: 0, q: 0 };
        const currentB = { p: 0, n: 0, b: 0, r: 0, q: 0 };
        const board = chess.board();
        for (let r = 0; r < 8; r++) {
            for (let c = 0; c < 8; c++) {
                const p = board[r][c];
                if (p && p.type !== 'k') {
                    if (p.color === 'w') currentW[p.type]++;
                    else currentB[p.type]++;
                }
            }
        }
        const icons = { p: '♟', n: '♞', b: '♝', r: '♜', q: '♛' };
        const vals = { p: 1, n: 3, b: 3, r: 5, q: 9 };
        let wCapStr = '', bCapStr = '', wMat = 0, bMat = 0;
        for (const t of ['q', 'r', 'b', 'n', 'p']) {
            const lostB = Math.max(0, startCounts[t] - currentB[t]);
            const lostW = Math.max(0, startCounts[t] - currentW[t]);
            wCapStr += icons[t].repeat(lostB);
            bCapStr += icons[t].repeat(lostW);
            wMat += currentW[t] * vals[t];
            bMat += currentB[t] * vals[t];
        }
        document.getElementById('playerCapturedList').textContent = wCapStr;
        document.getElementById('oppCapturedList').textContent = bCapStr;
        const diff = wMat - bMat;
        document.getElementById('materialAdvantage').textContent = diff > 0 ? `+${diff}` : '';
    }

    let peer = null;

    function initPeerJS() {
        if (typeof Peer === 'undefined') return;
        try {
            const cleanId = 'krypta-' + currentUser.username.replace(/[^a-zA-Z0-9]/g, '') + '-' + Math.floor(100 + Math.random() * 899);
            peer = new Peer(cleanId);
            peer.on('open', id => {

                // Sprawdź parametr ?invite= w URL
                const params = new URLSearchParams(window.location.search);
                const inviteId = params.get('invite');
                if (inviteId) {
                    selectOpponentMode('p2p');
                    document.getElementById('remotePeerInput').value = inviteId;
                    connectToPeer();
                }
            });
            peer.on('connection', c => {
                conn = c;
                setupPeerConnection();
                showToast('Znajomy połączył się do Twojej Krypty P2P!', 'success');
            });
        } catch (e) {}
    }

    function connectToPeer() {
        const targetId = document.getElementById('remotePeerInput').value.trim();
        if (!peer || !targetId) return;
        conn = peer.connect(targetId);
        setupPeerConnection();
        showToast('Łączenie z rywalem P2P...', 'info');
    }

    function setupPeerConnection() {
        conn.on('data', data => {
            if (data.type === 'move' && gameState.type === 'chess') {
                const fromAlg = coordsToAlg(data.fromR, data.fromC);
                const toAlg = coordsToAlg(data.toR, data.toC);
                chess.move({ from: fromAlg, to: toAlg, promotion: data.promo || 'q' });
                gameState.lastMove = { fromR: data.fromR, fromC: data.fromC, toR: data.toR, toC: data.toC };
                audio.playMove('move');
                updateMoveHistoryUI();
                renderBoard();
                checkChessEndConditions();
            } else if (data.type === 'chat') {
                appendChatMessage(data.sender, data.text);
            }
        });
    }

    // Tło cząsteczkowe (pył krypty / iskry zsynchronizowane z porą dnia)
    function initParticles() {
        const canvas = document.getElementById('particleCanvas');
        const ctx = canvas.getContext('2d');
        let w = canvas.width = window.innerWidth;
        let h = canvas.height = window.innerHeight;
        window.addEventListener('resize', () => {
            w = canvas.width = window.innerWidth;
            h = canvas.height = window.innerHeight;
        });

        const hour = new Date().getHours();
        const isNight = hour >= 19 || hour <= 6;
        const particles = Array.from({ length: 55 }, () => ({
            x: Math.random() * w,
            y: Math.random() * h,
            r: Math.random() * 2 + 0.6,
            vx: (Math.random() - 0.5) * 0.4,
            vy: -Math.random() * 0.5 - 0.15,
            alpha: Math.random() * 0.6 + 0.1
        }));

        function animate() {
            ctx.clearRect(0, 0, w, h);
            if (!currentUser || currentUser.settings.particles !== false) {
                for (const p of particles) {
                    p.x += p.vx;
                    p.y += p.vy;
                    if (p.y < 0) p.y = h;
                    if (p.x < 0) p.x = w;
                    if (p.x > w) p.x = 0;

                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
                    ctx.fillStyle = isNight ? `rgba(212, 175, 55, ${p.alpha})` : `rgba(148, 163, 184, ${p.alpha})`;
                    ctx.fill();
                }
