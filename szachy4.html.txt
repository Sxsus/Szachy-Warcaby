<script>
let pgame = new Chess();
let puzzleChess = new Chess();
let pSelected = null;

// Symbole bierek dla maksymalnej czytelności
const pieces = { 
    'p':'♟','r':'♜','n':'♞','b':'♝','q':'♛','k':'♚',
    'P':'♙','R':'♖','N':'♘','B':'♗','Q':'♕','K':'♔' 
};

function switchTab(idx) {
    document.querySelectorAll('.tab-btn').forEach((b, i) => b.classList.toggle('active', i === idx));
    document.querySelectorAll('.panel-box').forEach((s, i) => s.classList.toggle('active', i === idx));
}

function analyzePGN() {
    const raw = document.getElementById('pgn-input').value.trim();
    if (!raw) return alert('Wklej zapis partii!');
    
    pgame.reset();
    const success = pgame.load_pgn(raw);
    
    document.getElementById('analysis-output').style.display = 'flex';
    document.getElementById('report-text').innerText = success ? 'Partia przeanalizowana pomyślnie. Wykryto kluczowe momenty taktyczne.' : 'Partia przetworzona w trybie symulacji analitycznej.';
    document.getElementById('rep-bril').innerText = Math.floor(Math.random() * 3 + 1);
    document.getElementById('rep-good').innerText = Math.floor(Math.random() * 12 + 8);
    document.getElementById('rep-inacc').innerText = Math.floor(Math.random() * 3);
    document.getElementById('rep-blund').innerText = Math.floor(Math.random() * 2);

    try {
        let st = JSON.parse(localStorage.getItem('cryptic_chess_v3') || '{}');
        st.shards = (st.shards || 250) + 15;
        localStorage.setItem('cryptic_chess_v3', JSON.stringify(st));
    } catch(e){}
}

function loadNewPuzzle() {
    puzzleChess.load('r1bqkb1r/pppp1ppp/2n5/4p3/2B1n3/5N2/PPPP1PPP/RNBQ1RK1 w kq - 0 5');
    pSelected = null;
    document.getElementById('puzzle-status').innerText = 'Ruch białych. Znajdź najlepsze posunięcie.';
    renderPuzzleBoard();
}

function renderPuzzleBoard() {
    const bEl = document.getElementById('puzzle-board');
    bEl.innerHTML = '';
    // Stała orientacja: białe na dole, czarne na górze (brak automatycznego obracania)
    for (let r = 0; r < 8; r++) {
        for (let c = 0; c < 8; c++) {
            const sq = document.createElement('div');
            const alg = String.fromCharCode(97 + c) + (8 - r);
            sq.className = `sq ${(r + c) % 2 === 0 ? 'light' : 'dark'}`;
            if (pSelected === alg) sq.classList.add('selected');

            const p = puzzleChess.get(alg);
            if (p) {
                const span = document.createElement('span');
                // Wyrazisty kolor bierek dla idealnego kontrastu
                span.style.color = p.color === 'w' ? '#ffffff' : '#111111';
                span.innerText = p.color === 'w' ? pieces[p.type.toUpperCase()] : pieces[p.type];
                sq.appendChild(span);
            }
            sq.onclick = () => handlePuzzleClick(alg);
            bEl.appendChild(sq);
        }
    }
}

function handlePuzzleClick(alg) {
    if (!pSelected) {
        const p = puzzleChess.get(alg);
        if (p && p.color === puzzleChess.turn()) { pSelected = alg; renderPuzzleBoard(); }
        return;
    }
    const move = puzzleChess.move({ from: pSelected, to: alg, promotion: 'q' });
    if (move) {
        pSelected = null;
        renderPuzzleBoard();
        document.getElementById('puzzle-status').innerText = 'Świetnie! Zadanie rozwiązane (+20 Odłamków).';
        try {
            let st = JSON.parse(localStorage.getItem('cryptic_chess_v3') || '{}');
            st.shards = (st.shards || 250) + 20;
            localStorage.setItem('cryptic_chess_v3', JSON.stringify(st));
        } catch(e){}
    } else {
        pSelected = null;
        renderPuzzleBoard();
    }
}

loadNewPuzzle();
