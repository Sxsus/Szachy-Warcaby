<script>
let chess = new Chess();
let selectedSq = null;
let draggedFrom = null;
let boardFlipped = false;
let currentBot = 'kokoszka';
let isMobileDevice = false;

const pieceSymbols = { 'p':'♟','r':'♜','n':'♞','b':'♝','q':'♛','k':'♚','P':'♙','R':'♖','N':'♘','B':'♗','Q':'♕','K':'♔' };

const bots = {
    'kokoszka': { name: 'Kokoszka', welcome: 'Ko-ko! Dziobiem w centrum szachownicy!' },
    'fischer': { name: 'Bobby Fischer', welcome: 'Gramy ostro i bezkompromisowo.' },
    'magnus': { name: 'Magnus Carlsen', welcome: 'Skupienie i precyzja na każdym ekranie.' },
    'hikaru': { name: 'Hikaru Nakamura', welcome: 'Szybkie tempo gry!' }
};

// Automatyczne wykrywanie urządzenia (Mobilne vs PC)
function detectDeviceType() {
    const ua = navigator.userAgent;
    const isTouch = (('ontouchstart' in window) || (navigator.maxTouchPoints > 0));
    const isSmallScreen = window.innerWidth <= 800;
    
    if (/Mobi|Android|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(ua) || (isTouch && isSmallScreen)) {
        isMobileDevice = true;
        document.getElementById('device-type-text').innerText = "Telefon / Tablet (Mobile Touch)";
    } else {
        isMobileDevice = false;
        document.getElementById('device-type-text').innerText = "Komputer (PC / Desktop)";
    }
}

function switchBot() {
    currentBot = document.getElementById('bot-selector').value;
    addChat(bots[currentBot].name + ": " + bots[currentBot].welcome, "coach");
}

function resetGame() {
    chess.reset();
    selectedSq = null;
    draggedFrom = null;
    renderBoard();
    updateEval();
    addChat("Trener: Nowa partia rozpoczęta!", "coach");
}

function renderBoard() {
    const boardEl = document.getElementById('chessboard');
    boardEl.innerHTML = '';
    const legalMoves = selectedSq ? chess.moves({ square: selectedSq, verbose: true }).map(m => m.to) : [];

    for (let i = 0; i < 64; i++) {
        const r = boardFlipped ? Math.floor(i / 8) : 7 - Math.floor(i / 8);
        const c = boardFlipped ? 7 - (i % 8) : i % 8;
        const alg = String.fromCharCode(97 + c) + (r + 1);

        const sq = document.createElement('div');
        sq.className = `sq ${(r + c) % 2 === 0 ? 'light' : 'dark'}`;
        if (selectedSq === alg) sq.classList.add('selected');
        if (legalMoves.includes(alg)) sq.classList.add('legal-target');

        const p = chess.get(alg);
        if (p) {
            if (p.type === 'k' && chess.in_check() && p.color === chess.turn()) {
                sq.classList.add('check-danger');
            }
            const span = document.createElement('span');
            span.style.color = p.color === 'w' ? '#ffffff' : '#111111';
            span.innerText = p.color === 'w' ? pieceSymbols[p.type.toUpperCase()] : pieceSymbols[p.type];
            sq.appendChild(span);

            // Jeśli to PC, włączamy wsparcie dla Drag & Drop
            if (!isMobileDevice) {
                sq.draggable = true;
                sq.addEventListener('dragstart', (e) => {
                    draggedFrom = alg;
                    e.dataTransfer.setData('text/plain', alg);
                });
            }
        }

        if (!isMobileDevice) {
            sq.addEventListener('dragover', (e) => e.preventDefault());
            sq.addEventListener('drop', (e) => {
                e.preventDefault();
                const from = draggedFrom || selectedSq;
                const to = alg;
                if (from) tryMove(from, to);
            });
        }

        sq.onclick = () => handleSquareClick(alg);
        boardEl.appendChild(sq);
    }
}

function handleSquareClick(alg) {
    if (!selectedSq) {
        const p = chess.get(alg);
        if (p && p.color === chess.turn()) {
            selectedSq = alg;
            renderBoard();
        }
        return;
    }
    tryMove(selectedSq, alg);
}

function tryMove(from, to) {
    const move = chess.move({ from: from, to: to, promotion: 'q' });
    if (move) {
        selectedSq = null;
        draggedFrom = null;
        renderBoard();
        updateEval();
        if (!chess.game_over() && chess.turn() === 'b') {
            setTimeout(botMove, 400);
        }
    } else {
        const p = chess.get(alg);
        if (p && p.color === chess.turn()) {
            selectedSq = alg;
        } else {
            selectedSq = null;
        }
        draggedFrom = null;
        renderBoard();
    }
}

function botMove() {
    const moves = chess.moves({ verbose: true });
    if (moves.length === 0) return;
    const chosen = moves[Math.floor(Math.random() * moves.length)];
    chess.move(chosen);
    renderBoard();
    updateEval();
}

function undoMove() {
    chess.undo();
    chess.undo();
    selectedSq = null;
    renderBoard();
    updateEval();
}

function flipBoard() {
    boardFlipped = !boardFlipped;
    renderBoard();
}

function resignGame() {
    if (confirm("Poddajesz partię?")) {
        addChat("Trener: Partia zakończona poddaniem.", "coach");
    }
}

function updateEval() {
    const val = 50 + (Math.random() * 10 - 5);
    document.getElementById('eval-fill').style.width = val + '%';
    document.getElementById('eval-score').innerText = ((val - 50) / 10).toFixed(1);
}

function sendMsg() {
    const input = document.getElementById('chat-input');
    const txt = input.value.trim();
    if (!txt) return;

    addChat("Szymon: " + txt, "player");
    input.value = "";

    setTimeout(() => {
        const deviceLabel = isMobileDevice ? "telefonie" : "komputerze";
        addChat(bots[currentBot].name + `: Świetny ruch na ${deviceLabel}! Gramy dalej.`, "coach");
    }, 500);
}

function addChat(text, type) {
    const logs = document.getElementById('chat-logs');
    const div = document.createElement('div');
    div.className = `msg-mob ${type}`;
    div.innerText = text;
    logs.appendChild(div);
    logs.scrollTop = logs.scrollHeight;
}

function initApp() {
    detectDeviceType();
    try {
        let raw = localStorage.getItem('cryptic_chess_v3');
        if (raw) {
            const st = JSON.parse(raw);
            if (st.settings && st.settings.lightTheme) document.documentElement.setAttribute('data-theme', 'light');
            document.getElementById('header-shards').innerText = st.shards ?? 250;
        }
    } catch(e){}
    renderBoard();
    updateEval();
}

initApp();
