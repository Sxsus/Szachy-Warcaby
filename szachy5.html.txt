function renderBoard() {
    const boardEl = document.getElementById('chessboard');
    boardEl.innerHTML = '';
    const legalMoves = selectedSq ? chess.moves({ square: selectedSq, verbose: true }).map(m => m.to) : [];

    for (let i = 0; i < 64; i++) {
        const r = boardFlipped ? Math.floor(i / 8) : 7 - Math.floor(i / 8);
        const c = boardFlipped ? 7 - (i % 8) : i % 8;
        const alg = String.fromCharCode(97 + c) + (r + 1);

        const sq = document.createElement('div');
        sq.className = `sq ${(r + c) % 2 === 0 ? 'light' : 'dark'}`;
        if (selectedSq === alg) sq.classList.add('selected');
        if (legalMoves.includes(alg)) sq.classList.add('legal-target');

        const p = chess.get(alg);
        if (p) {
            if (p.type === 'k' && chess.in_check() && p.color === chess.turn()) {
                sq.classList.add('check-danger');
            }
            const span = document.createElement('span');
            span.style.color = p.color === 'w' ? '#ffffff' : '#111111';
            span.innerText = p.color === 'w' ? pieceSymbols[p.type.toUpperCase()] : pieceSymbols[p.type];
            sq.appendChild(span);

            // Drag & Drop fizyka bierek
            sq.draggable = true;
            sq.addEventListener('dragstart', (e) => {
                draggedFrom = alg;
                e.dataTransfer.setData('text/plain', alg);
            });
        }

        sq.addEventListener('dragover', (e) => e.preventDefault());
        sq.addEventListener('drop', (e) => {
            e.preventDefault();
            const from = draggedFrom || selectedSq;
            const to = alg;
            if (from) tryMove(from, to);
        });

        sq.onclick = () => handleSquareClick(alg);
        boardEl.appendChild(sq);
    }
}

function handleSquareClick(alg) {
    if (!selectedSq) {
        const p = chess.get(alg);
        if (p && p.color === chess.turn()) {
            selectedSq = alg;
            renderBoard();
        }
        return;
    }
    tryMove(selectedSq, alg);
}

function tryMove(from, to) {
    const move = chess.move({ from: from, to: to, promotion: 'q' });
    if (move) {
        selectedSq = null;
        draggedFrom = null;
        renderBoard();
        updateHistory();
        updateEvalBar();
        
        if (soundEnabled) playClickSound();

        // Odpowiedź bota / Kokoszki
        if (!chess.game_over() && chess.turn() === 'b') {
            setTimeout(makeBotMove, 600);
        }
    } else {
        // AUTOMATYCZNE COFNIĘCIE BŁĘDNEGO RUCHU
        selectedSq = null;
        draggedFrom = null;
        renderBoard();
        const botObj = botDatabase[currentBotKey];
        let errText = botObj.name + ": Ten ruch jest niedozwolony! Bierka wraca na pole.";
        if (currentBotKey === 'kokoszka') errText = "Kokoszka (Ruda Kura): Ko-ko-korrr! Tak nie wolno skakać! Dziobiem w poprawny ruch!";
        addChatMessage(errText, "coach");
    }
}

function makeBotMove() {
    const moves = chess.moves({ verbose: true });
    if (moves.length === 0) return;
    const chosen = moves[Math.floor(Math.random() * moves.length)];
    chess.move(chosen);
    renderBoard();
    updateHistory();
    updateEvalBar();

    const botObj = botDatabase[currentBotKey];
    let comment = "Ciekawe posunięcie, gramy dalej w centrum.";
    if (currentBotKey === 'kokoszka') {
        const kComments = ["Ko-ko! Znoszę jajko na polu d5!", "Dziob w Twojego skoczka!", "Ko-ko-korrr! Twój król gdacze ze strachu!"];
        comment = kComments[Math.floor(Math.random() * kComments.length)];
    } else if (currentBotKey === 'fischer') {
        comment = "Fischer: Otwarta pozycja premiuje atak. Liczy się tylko wygrana.";
    }
    addChatMessage(botObj.name + ": " + comment, "coach");
}

function updateHistory() {
    const history = chess.history();
    const box = document.getElementById('history-box');
    if (history.length === 0) {
        box.innerText = "Brak rozegranych posunięć w tej sesji.";
        return;
    }
    let text = "";
    for (let i = 0; i < history.length; i += 2) {
        text += `${Math.floor(i/2)+1}. ${history[i]} ${history[i+1] || ''} \n`;
    }
    box.innerText = text;
    box.scrollTop = box.scrollHeight;
}

function playClickSound() {
    try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.frequency.setValueAtTime(440, ctx.currentTime);
        gain.gain.setValueAtTime(0.04, ctx.currentTime);
        osc.start();
        osc.stop(ctx.currentTime + 0.08);
    } catch(e){}
}
