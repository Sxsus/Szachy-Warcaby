<script>
let board = [];
let turn = 'w';
let selectedSq = null;
let matchActive = false;
let isP2P = false;
let peer = null, conn = null;
let timeW = 600, timeB = 600, timer = null;
let disconnectTimer = null;

function syncTheme() {
    try {
        const raw = localStorage.getItem('cryptic_chess_v3');
        if (raw) {
            const st = JSON.parse(raw);
            if (st.settings && st.settings.lightTheme) document.documentElement.setAttribute('data-theme', 'light');
            document.getElementById('me-name').innerText = st.nick || 'Szymon';
        }
    } catch(e){}
}

function toggleP2PInput(val) {
    isP2P = (val === 'p2p');
    document.getElementById('p2p-box').style.display = isP2P ? 'flex' : 'none';
    if (isP2P && !peer) initPeer();
}

function initPeer() {
    peer = new Peer();
    peer.on('open', id => {
        document.getElementById('my-peer-id').innerText = id;
    });
    peer.on('connection', c => {
        conn = c;
        setupConnection();
        logChat("P2P: Połączono z graczem!");
    });
}

function connectToPeer() {
    const id = document.getElementById('friend-peer-id').value.trim();
    if (!id) return alert('Wpisz ID znajomego!');
    conn = peer.connect(id);
    setupConnection();
}

function setupConnection() {
    conn.on('data', data => {
        if (data.type === 'move') {
            board = data.board;
            turn = 'w';
            renderCheckersBoard();
        } else if (data.type === 'chat') {
            logChat("Przeciwnik: " + data.text);
        } else if (data.type === 'resign') {
            endCheckersMatch(true, 'Przeciwnik poddał się!');
        }
    });
}

function startCheckersMatch() {
    syncTheme();
    matchActive = true;
    turn = 'w';
    selectedSq = null;
    timeW = 600; timeB = 600;

    board = Array(8).fill(null).map(() => Array(8).fill(null));
    for (let r = 0; r < 8; r++) {
        for (let c = 0; c < 8; c++) {
            if ((r + c) % 2 === 1) {
                if (r < 3) board[r][c] = { color: 'b', king: false };
                if (r > 4) board[r][c] = { color: 'w', king: false };
            }
        }
    }

    document.getElementById('setup-screen').style.display = 'none';
    document.getElementById('game-screen').style.display = 'flex';
    document.getElementById('checkers-chat').innerHTML = '';
    
    logChat("Bitwa warcabowa rozpoczęta! Pilnuj zasady dmuchnięcia.");
    renderCheckersBoard();
    startCheckersClock();
}

function getAllCaptures(color) {
    let caps = [];
    for (let r = 0; r < 8; r++) {
        for (let c = 0; c < 8; c++) {
            const p = board[r][c];
            if (p && p.color === color) {
                const dirs = [[1,1],[1,-1],[-1,1],[-1,-1]];
                for (let [dr, dc] of dirs) {
                    if (!p.king) {
                        const mr = r + dr, mc = c + dc, lr = r + 2*dr, lc = c + 2*dc;
                        if (lr >= 0 && lr < 8 && lc >= 0 && lc < 8) {
                            if (board[mr][mc] && board[mr][mc].color !== color && !board[lr][lc]) {
                                caps.push({ from: [r,c], to: [lr,lc], mid: [mr,mc] });
                            }
                        }
                    } else {
                        let step = 1, enemy = null;
                        while (r + step*dr >= 0 && r + step*dr < 8 && c + step*dc >= 0 && c + step*dc < 8) {
                            const cr = r + step*dr, cc = c + step*dc;
                            const cell = board[cr][cc];
                            if (!cell) {
                                if (enemy) caps.push({ from: [r,c], to: [cr,cc], mid: enemy });
                            } else if (cell.color === color) {
                                break;
                            } else {
                                if (enemy) break;
                                enemy = [cr, cc];
                            }
                            step++;
                        }
                    }
                }
            }
        }
    }
    return caps;
}

function renderCheckersBoard() {
    const bEl = document.getElementById('checkers-board');
    bEl.innerHTML = '';
    for (let r = 0; r < 8; r++) {
        for (let c = 0; c < 8; c++) {
            const sq = document.createElement('div');
            sq.className = `sq ${(r + c) % 2 === 0 ? 'light' : 'dark'}`;
            if (selectedSq && selectedSq[0] === r && selectedSq[1] === c) sq.classList.add('selected');

            const p = board[r][c];
            if (p) {
                const span = document.createElement('span');
                span.innerText = p.color === 'w' ? (p.king ? '👑' : '⚪') : (p.king ? '👹' : '⚫');
                sq.appendChild(span);
            }
            sq.onclick = () => handleCheckersClick(r, c);
            bEl.appendChild(sq);
        }
    }
}

function handleCheckersClick(r, c) {
    if (!matchActive || turn !== 'w') return;
    const clicked = board[r][c];
    if (clicked && clicked.color === 'w') {
        selectedSq = [r, c];
        renderCheckersBoard();
        return;
    }
    if (!selectedSq) return;

    const [sr, sc] = selectedSq;
    const piece = board[sr][sc];
    const caps = getAllCaptures('w');
    const chosenCap = caps.find(cp => cp.from[0] === sr && cp.from[1] === sc && cp.to[0] === r && cp.to[1] === c);

    if (chosenCap) {
        board[r][c] = piece;
        board[sr][sc] = null;
        board[chosenCap.mid[0]][chosenCap.mid[1]] = null;
        if (r === 0) piece.king = true;
        if (window.OptimizedAudio) window.OptimizedAudio.playClick(180, 'sawtooth');
        selectedSq = null;
        turn = 'b';
        renderCheckersBoard();
        if (isP2P && conn) conn.send({ type: 'move', board });
        else setTimeout(botCheckersMove, 600);
    } else {
        const dr = r - sr, dc = c - sc;
        const valid = !piece.king ? (dr === -1 && Math.abs(dc) === 1 && !board[r][c]) : (Math.abs(dr) === Math.abs(dc) && !board[r][c]);
        if (valid) {
            board[r][c] = piece;
            board[sr][sc] = null;
            if (r === 0) piece.king = true;

            // ZASADA DMUCHNIĘCIA
            if (caps.length > 0) {
                const target = caps[0].from;
                board[target[0]][target[1]] = null;
                logChat("💨 DMUCHNIĘCIE! Zignorowano przymusowe bicie i tracisz pionka!");
                if (window.OptimizedAudio) window.OptimizedAudio.playClick(100, 'square');
            } else {
                if (window.OptimizedAudio) window.OptimizedAudio.playClick(320);
            }
            selectedSq = null;
            turn = 'b';
            renderCheckersBoard();
            if (isP2P && conn) conn.send({ type: 'move', board });
            else setTimeout(botCheckersMove, 600);
        }
    }
}

function botCheckersMove() {
    if (!matchActive) return;
    const caps = getAllCaptures('b');
    if (caps.length > 0) {
        const cp = caps[0];
        const p = board[cp.from[0]][cp.from[1]];
        board[cp.to[0]][cp.to[1]] = p;
        board[cp.from[0]][cp.from[1]] = null;
        board[cp.mid[0]][cp.mid[1]] = null;
        if (cp.to[0] === 7) p.king = true;
    } else {
        for (let r = 0; r < 8; r++) {
            for (let c = 0; c < 8; c++) {
                const p = board[r][c];
                if (p && p.color === 'b') {
                    for (let dc of [-1, 1]) {
                        if (r + 1 < 8 && c + dc >= 0 && c + dc < 8 && !board[r+1][c+dc]) {
                            board[r+1][c+dc] = p;
                            board[r][c] = null;
                            if (r + 1 === 7) p.king = true;
                            turn = 'w';
                            renderCheckersBoard();
                            return;
                        }
                    }
                }
            }
        }
    }
    turn = 'w';
    renderCheckersBoard();
}

function startCheckersClock() {
    clearInterval(timer);
    timer = setInterval(() => {
        if (!matchActive) return;
        if (turn === 'w') timeW--; else timeB--;
        document.getElementById('clock-white').innerText = `${Math.floor(timeW/60)}:${String(timeW%60).padStart(2,'0')}`;
        document.getElementById('clock-black').innerText = `${Math.floor(timeB/60)}:${String(timeB%60).padStart(2,'0')}`;
        if (timeW <= 0) endCheckersMatch(false, 'Koniec czasu!');
        if (timeB <= 0) endCheckersMatch(true, 'Czas przeciwnika minął!');
    }, 1000);
}

function simulateDisconnectTest() {
    logChat("⚠️ Symulacja rozłączenia. Oczekiwanie 15 sekund na powrót...");
    let left = 15;
    disconnectTimer = setInterval(() => {
        left--;
        if (left <= 0 || !matchActive) {
            clearInterval(disconnectTimer);
            if (matchActive) endCheckersMatch(true, 'Walkower (Przeciwnik opuścił grę na 15s)');
        }
    }, 1000);
}

function endCheckersMatch(won, reason) {
    matchActive = false;
    clearInterval(timer);
    clearInterval(disconnectTimer);

    try {
        const raw = localStorage.getItem('cryptic_chess_v3');
        if (raw) {
            const st = JSON.parse(raw);
            st.elo.checkersBot += (won === true ? 16 : (won === false ? -12 : 0));
            if (won) st.shards += 30;
            localStorage.setItem('cryptic_chess_v3', JSON.stringify(st));
        }
    } catch(e){}

    document.getElementById('v-title').innerText = won ? `Zwycięstwo! (${reason})` : `Porażka (${reason})`;
    document.getElementById('v-sub').innerText = won ? 'Zdobyto +30 Odłamków Cienia!' : 'Bitwa przegrana.';
    document.getElementById('modal-verdict').style.display = 'flex';
}

function closeCheckersVerdict() {
    document.getElementById('modal-verdict').style.display = 'none';
    backToCheckersSetup();
}

function backToCheckersSetup() {
    matchActive = false;
    clearInterval(timer);
    document.getElementById('game-screen').style.display = 'none';
    document.getElementById('setup-screen').style.display = 'flex';
}

function resignCheckers() {
    if (isP2P && conn) conn.send({ type: 'resign' });
    endCheckersMatch(false, 'Poddanie partii');
}

function logChat(txt) {
    const log = document.getElementById('checkers-chat');
    const div = document.createElement('div');
    div.className = 'msg bot';
    div.innerText = txt;
    log.appendChild(div);
    log.scrollTop = log.scrollHeight;
}

function sendCheckersChat() {
    const inp = document.getElementById('checkers-chat-txt');
    if (!inp.value.trim()) return;
    const log = document.getElementById('checkers-chat');
    const div = document.createElement('div');
    div.className = 'msg user';
    div.innerText = inp.value;
    log.appendChild(div);
    if (isP2P && conn) conn.send({ type: 'chat', text: inp.value });
    inp.value = '';
    log.scrollTop = log.scrollHeight;
}
