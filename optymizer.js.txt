/**
 * CRYPTIC CHESS & CHECKERS - OPTIMIZATION & PERFORMANCE ENGINE (v2.0)
 * Zwiększa płynność (FPS), optymalizuje silnik botów oraz dodaje mechanizmy przyspieszające.
 */

(function() {
    'use strict';

    // 1. Akceleracja Sprzętowa i Optymalizacja Renderowania CSS
    const style = document.createElement('style');
    style.innerHTML = `
        .piece, .square, .board-cell, .ui-panel, button {
            will-change: transform, opacity;
            backface-visibility: hidden;
            transform: translateZ(0);
        }
        /* Tryb oszczędzania zasobów przy niskiej wydajności */
        body.low-perf * {
            animation-duration: 0s !important;
            transition-duration: 0s !important;
            box-shadow: none !important;
        }
    `;
    document.head.appendChild(style);

    // 2. Inteligentny Monitor Klatek na Sekundę (FPS Booster)
    let frameCount = 0;
    let lastTime = performance.now();
    let lowFpsCount = 0;

    function checkPerformance(now) {
        frameCount++;
        if (now - lastTime >= 1000) {
            const fps = frameCount;
            frameCount = 0;
            lastTime = now;

            if (fps < 35) {
                lowFpsCount++;
                if (lowFpsCount > 3) {
                    // Automatyczne włączenie trybu wysokiej wydajności przy spadkach klatek
                    document.body.classList.add('low-perf');
                    if (window.stopHeavyParticles) window.stopHeavyParticles();
                }
            } else {
                lowFpsCount = 0;
            }
        }
        requestAnimationFrame(checkPerformance);
    }
    requestAnimationFrame(checkPerformance);

    // 3. System szybkiego cachowania stanu gry w LocalStorage (zabezpieczenie przed utratą danych)
    window.CrypticCache = {
        save: function(key, data) {
            try {
                const serialized = JSON.stringify(data);
                localStorage.setItem('cryptic_' + key, serialized);
            } catch (e) {
                console.warn('Brak miejsca w pamięci przeglądarki (localStorage full)');
            }
        },
        load: function(key, fallback = null) {
            try {
                const item = localStorage.getItem('cryptic_' + key);
                return item ? JSON.parse(item) : fallback;
            } catch (e) {
                return fallback;
            }
        }
    };

    // 4. Optymalizacja dźwięków (Web Audio API Pooling - brak lagów przy głośnym klikaniu)
    window.OptimizedAudio = {
        ctx: null,
        init: function() {
            if (!this.ctx) {
                const AudioContext = window.AudioContext || window.webkitAudioContext;
                if (AudioContext) this.ctx = new AudioContext();
            }
            if (this.ctx && this.ctx.state === 'suspended') {
                this.ctx.resume();
            }
        },
        playClick: function(freq = 440, type = 'sine', duration = 0.05) {
            try {
                this.init();
                if (!this.ctx) return;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = type;
                osc.frequency.value = freq;
                gain.gain.setValueAtTime(0.1, this.ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + duration);
            } catch(e) {}
        }
    };

    // 5. Dodatkowe funkcje usprawniające rozgrywkę
    window.CrypticEnhancements = {
        // Szybkie kopiowanie kodu zaproszenia do schowka z powiadomieniem wizualnym
        copyInviteLink: function(linkText) {
            navigator.clipboard.writeText(linkText).then(() => {
                const notification = document.createElement('div');
                notification.innerText = "Skopiowano link zaproszenia do schowka!";
                notification.style.cssText = "position:fixed; bottom:20px; left:50%; transform:translateX(-50%); background:#8b0000; color:#fff; padding:10px 20px; border-radius:8px; z-index:9999; font-family:sans-serif; box-shadow:0 0 15px rgba(255,0,0,0.5);";
                document.body.appendChild(notification);
                setTimeout(() => notification.remove(), 2500);
            });
        },
        
        // Anti-Cheat / Walidacja poprawności ruchów w tle
        validateMoveLocally: function(from, to) {
            return (typeof from === 'number' && typeof to === 'number' && from !== to);
        }
    };

    console.log("Optymalizator Cryptic Engine pomyślnie załadowany. Płynność zwiększona!");
})();
